cmake_minimum_required(VERSION 3.16)
project(wasteland_engine CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(fod STATIC
    src/lzss.cpp src/archive.cpp src/map.cpp src/script.cpp src/gfx.cpp src/data.cpp src/state.cpp src/world.cpp src/vm.cpp src/screen.cpp src/game.cpp src/combat.cpp src/combat_fight.cpp src/progress.cpp src/shop.cpp src/party.cpp src/charsheet.cpp src/savegame.cpp src/creator.cpp src/sound.cpp src/timed.cpp)
target_include_directories(fod PUBLIC include)
target_compile_options(fod PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra>)

add_executable(fodrun tools/fodtool.cpp)
target_link_libraries(fodrun fod)
if(MINGW)
    # fully static executables (the system antivirus removes dynamically linked builds here)
    target_link_options(fodrun PRIVATE -static)
endif()
if(WIN32)
    add_executable(fodplay platform/win32_main.cpp)
    target_link_libraries(fodplay fod winmm)
    target_link_options(fodplay PRIVATE -static -mwindows)
endif()

add_library(wl STATIC wl/codec.cpp wl/gamefile.cpp wl/graphics.cpp wl/exe.cpp wl/machine.cpp wl/ui.cpp wl/video.cpp wl/assets.cpp wl/mapview.cpp wl/rules.cpp wl/party.cpp wl/text.cpp wl/x86.cpp)
target_include_directories(wl PUBLIC include)
target_compile_options(wl PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra>)
target_link_libraries(wl fod)

add_executable(wlrun tools/wltool.cpp)
target_link_libraries(wlrun wl)
if(MINGW)
    target_link_options(wlrun PRIVATE -static)
endif()

# development oracle: 8086 interpreter running the original WL.EXE image (not part of the engine)
add_executable(wlemu tools/wlemu.cpp)
target_link_libraries(wlemu wl)
if(MINGW)
    target_link_options(wlemu PRIVATE -static)
endif()

add_executable(wltest tools/wltest.cpp)
target_link_libraries(wltest wl)
if(MINGW)
    target_link_options(wltest PRIVATE -static)
endif()

# Statically recompiled WL.EXE. The image comes from WL.EXE (set -DWL_EXE=path/to/WL.EXE; unpacked and combined with
# WLA.BIN by wlimage) or from an image exported by ida_scripts/export_image.py. Nothing derived from the original program
# is committed: the generated C++ is a build artifact.
set(WL_EXE "" CACHE FILEPATH "path of the original WL.EXE")
set(WL_GAME_DIR ${CMAKE_SOURCE_DIR}/../wasteland1/game CACHE PATH "directory with WLA.BIN and the data files")
add_executable(wlimage tools/wlimage.cpp)
target_link_libraries(wlimage wl)
set(WL_IMAGE ${CMAKE_SOURCE_DIR}/../ida_scripts/wl_image.bin)
if(WL_EXE AND EXISTS ${WL_EXE})
    set(WL_IMAGE ${CMAKE_BINARY_DIR}/wl_image.bin)
    add_custom_command(OUTPUT ${WL_IMAGE}
        COMMAND wlimage ${WL_EXE} ${WL_GAME_DIR}/WLA.BIN ${WL_IMAGE}
        DEPENDS wlimage ${WL_EXE} ${WL_GAME_DIR}/WLA.BIN)
    set(WL_HAVE_IMAGE TRUE)
elseif(EXISTS ${WL_IMAGE})
    set(WL_HAVE_IMAGE TRUE)
endif()
if(WL_HAVE_IMAGE)
    option(WL_RECOMP_CHECK "wrap recompiled functions with a stack balance check" OFF)
    set(WL_CHECK_ARG "")
    if(WL_RECOMP_CHECK)
        set(WL_CHECK_ARG --check)
    endif()
    set(WL_GEN ${CMAKE_BINARY_DIR}/wl_gen.cpp)
    add_custom_command(OUTPUT ${WL_GEN}
        COMMAND python ${CMAKE_SOURCE_DIR}/tools/recomp/recomp.py ${WL_IMAGE} ${CMAKE_SOURCE_DIR}/tools/recomp/entries.txt ${WL_GEN} --interp 6C9,6CE,4330,4388,43A8,43D0 ${WL_CHECK_ARG}
        DEPENDS ${CMAKE_SOURCE_DIR}/tools/recomp/recomp.py ${CMAKE_SOURCE_DIR}/tools/recomp/entries.txt ${WL_IMAGE})
    add_library(wlgen STATIC ${WL_GEN} wl/recomp_rt.cpp)
    target_link_libraries(wlgen wl)
    add_executable(wlrc tools/wlrc.cpp)
    target_link_libraries(wlrc wlgen wl)
    if(MINGW)
        target_link_options(wlrc PRIVATE -static)
    endif()
    add_executable(wlrctest tools/wlrctest.cpp)
    target_link_libraries(wlrctest wlgen wl)
    if(MINGW)
        target_link_options(wlrctest PRIVATE -static)
    endif()
    if(WIN32)
        add_executable(wlplay platform/win32_wl.cpp)
        target_link_libraries(wlplay wlgen wl)
        target_link_options(wlplay PRIVATE -static -mwindows)
    endif()
endif()
